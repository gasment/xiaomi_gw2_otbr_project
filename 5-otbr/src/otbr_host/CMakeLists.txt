cmake_minimum_required(VERSION 3.16)
project(mi_otbr_host LANGUAGES C CXX)
# VERSION belongs to the enclosing project, independent of its directory name.
set(MI_OTBR_VERSION_FILE "${CMAKE_CURRENT_SOURCE_DIR}/../../../VERSION")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${MI_OTBR_VERSION_FILE}")
file(READ "${MI_OTBR_VERSION_FILE}" MI_OTBR_PROJECT_VERSION)
string(STRIP "${MI_OTBR_PROJECT_VERSION}" MI_OTBR_PROJECT_VERSION)
if(NOT MI_OTBR_PROJECT_VERSION MATCHES "^[0-9]+[.][0-9]+[.][0-9]+[A-Za-z0-9.+-]*$")
  message(FATAL_ERROR "Invalid project VERSION")
endif()
set(CMAKE_C_STANDARD 99)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

option(OTBR_HOST_STATIC "Link standalone tools statically" OFF)
option(OTBR_HOST_BUILD_API "Build independent Native API protocol check server" OFF)
option(OTBR_HOST_BUILD_RUNTIME "Build combined OTBR + Native API runtime" OFF)
option(OTBR_HOST_BUILD_WEB "Build standalone HTTP transport check" OFF)
option(OTBR_HOST_KERNEL_MROUTE "Kernel supports IPv6 multicast routing (BBR forwarding)" OFF)
option(OTBR_HOST_USERSPACE_MROUTE "Experimental Linux AF_PACKET BBR forwarding" OFF)

  set(OTBR_HOST_BUILD_API ON)
  set(OTBR_HOST_BUILD_WEB ON)

if(OTBR_HOST_BUILD_API)
  if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/vendor/noise-c/CMakeLists.txt")
    message(FATAL_ERROR "Run sh prepare-api.sh before building Native API.")
  endif()
  set(NOISE_C_BUILD_TESTS OFF CACHE BOOL "" FORCE)
  set(NOISE_C_FIND_LIBSODIUM OFF CACHE BOOL "" FORCE)
  add_subdirectory(vendor/noise-c EXCLUDE_FROM_ALL)
  target_compile_definitions(noise_c PUBLIC
    NOISE_USE_REFERENCE_BACKEND=1 NOISE_USE_LIBSODIUM=0 NOISE_USE_OPENSSL=0
    NOISE_USE_CUSTOM_RAND=1 NOISE_USE_PROTOCOL_NAME_TABLE=0 NOISE_USE_FALLBACK=0 NOISE_USE_HFS=0)
  # Keep the reference backend's internal Donna API separate from OpenSSL's
  # ARM assembler exports. Apply to both definitions and all Noise callers.
  target_compile_definitions(noise_c PRIVATE
    poly1305_init=mi_noise_poly1305_init
    poly1305_update=mi_noise_poly1305_update
    poly1305_finish=mi_noise_poly1305_finish
    poly1305_auth=mi_noise_poly1305_auth
    poly1305_verify=mi_noise_poly1305_verify
    poly1305_power_on_self_test=mi_noise_poly1305_power_on_self_test)
  target_sources(noise_c PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/src/noise_random.cpp")
  add_library(otbr-host-native-api STATIC src/native_api.cpp)
  target_link_libraries(otbr-host-native-api PUBLIC otbr-host-config esphome::noise_c Threads::Threads)
  target_compile_options(otbr-host-native-api PRIVATE -Wall -Wextra -Wpedantic)
  add_executable(otbr-native-api-check src/api_check_main.cpp)
  target_link_libraries(otbr-native-api-check PRIVATE otbr-host-native-api)
  if(OTBR_HOST_STATIC)
    target_link_options(otbr-native-api-check PRIVATE -static)
  endif()
endif()
if(OTBR_HOST_BUILD_WEB)
  set(MI_WEB_VENDOR "${CMAKE_CURRENT_SOURCE_DIR}/vendor/ot-br-posix/third_party")
  if(NOT EXISTS "${MI_WEB_VENDOR}/cpp-httplib/repo/httplib.h" OR NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/web/generated/web_assets.hpp")
    message(FATAL_ERROR "Run sh prepare-web.sh before building Web.")
  endif()
  add_library(otbr-host-web STATIC src/web.cpp)
  target_include_directories(otbr-host-web PRIVATE "${MI_WEB_VENDOR}/cpp-httplib/repo" web/generated)
  target_link_libraries(otbr-host-web PUBLIC otbr-host-config Threads::Threads)
  target_compile_options(otbr-host-web PRIVATE -Wall -Wextra -Wpedantic)
  if(CMAKE_CROSSCOMPILING)
    set(MI_SSL_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../deps/openssl-arm")
    if(NOT EXISTS "${MI_SSL_ROOT}/lib/libssl.a")
      message(FATAL_ERROR "Missing existing static ARM OpenSSL in deps/openssl-arm")
    endif()
    target_include_directories(otbr-host-web PUBLIC "${MI_SSL_ROOT}/include")
    target_link_libraries(otbr-host-web PUBLIC "${MI_SSL_ROOT}/lib/libssl.a" "${MI_SSL_ROOT}/lib/libcrypto.a" ${CMAKE_DL_LIBS})
  else()
    find_package(OpenSSL REQUIRED)
    target_link_libraries(otbr-host-web PUBLIC OpenSSL::SSL OpenSSL::Crypto)
  endif()
  target_compile_definitions(otbr-host-web PUBLIC CPPHTTPLIB_OPENSSL_SUPPORT)
  add_library(otbr-host-json STATIC "${MI_WEB_VENDOR}/cJSON/repo/cJSON.c")
  target_include_directories(otbr-host-json PUBLIC "${MI_WEB_VENDOR}/cJSON/repo")
  add_library(otbr-host-gateway-system STATIC src/gateway_system.cpp)
  target_link_libraries(otbr-host-gateway-system PUBLIC otbr-host-config otbr-host-json Threads::Threads)
  target_compile_options(otbr-host-gateway-system PRIVATE -Wall -Wextra -Wpedantic)
  add_executable(gateway-system-test tests/gateway_system_test.cpp)
  target_link_libraries(gateway-system-test PRIVATE otbr-host-gateway-system)
  if(OTBR_HOST_STATIC)
    target_link_options(gateway-system-test PRIVATE -static)
  endif()
  add_library(otbr-host-ha STATIC src/ha.cpp)
  target_include_directories(otbr-host-ha PRIVATE "${MI_WEB_VENDOR}/cpp-httplib/repo")
  target_link_libraries(otbr-host-ha PUBLIC otbr-host-web otbr-host-json)
  target_compile_options(otbr-host-ha PRIVATE -Wall -Wextra)
  add_executable(ha-client-check src/ha_check_main.cpp)
  target_link_libraries(ha-client-check PRIVATE otbr-host-ha)
  add_executable(otbr-web-http-check src/web_check_main.cpp)
  target_link_libraries(otbr-web-http-check PRIVATE otbr-host-web)
  add_executable(web-queue-test tests/web_queue_test.cpp)
  target_link_libraries(web-queue-test PRIVATE otbr-host-web)
  if(OTBR_HOST_STATIC)
    target_link_options(otbr-web-http-check PRIVATE -static)
    target_link_options(web-queue-test PRIVATE -static)
    target_link_options(ha-client-check PRIVATE -static)
  endif()
endif()
