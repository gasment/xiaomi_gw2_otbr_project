cmake_minimum_required(VERSION 3.16)
project(mi_otbr_core_baseline LANGUAGES C CXX)
get_filename_component(OTBR_HOST_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/.." ABSOLUTE)
option(MI_OTBR_STATIC "Link baseline executables statically" OFF)
option(MI_OTBR_KERNEL_MROUTE "Enable BBR multicast forwarding when supported by kernel" ON)
if(MI_OTBR_STATIC)
  string(APPEND CMAKE_EXE_LINKER_FLAGS " -static")
endif()
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(OTBR_DBUS OFF CACHE BOOL "" FORCE)
set(OTBR_WEB OFF CACHE BOOL "" FORCE)
set(OTBR_REST OFF CACHE BOOL "" FORCE)
set(OTBR_OPENWRT OFF CACHE BOOL "" FORCE)
set(OTBR_MDNS openthread CACHE STRING "" FORCE)
set(OTBR_BORDER_AGENT ON CACHE BOOL "" FORCE)
set(OTBR_BORDER_ROUTING ON CACHE BOOL "" FORCE)
option(MI_OTBR_USERSPACE_MROUTE "Experimental Linux userspace multicast backend" OFF)
if(MI_OTBR_KERNEL_MROUTE AND MI_OTBR_USERSPACE_MROUTE)
  message(FATAL_ERROR "Select kernel OR userspace multicast routing")
endif()
if(MI_OTBR_USERSPACE_MROUTE)
  # Enable BBR explicitly only after packet socket capability checks in Host.
  add_compile_definitions(OTBR_ENABLE_BACKBONE_ROUTER_ON_INIT=0)
endif()
if(MI_OTBR_KERNEL_MROUTE OR MI_OTBR_USERSPACE_MROUTE)
  set(OTBR_BACKBONE_ROUTER ON CACHE BOOL "" FORCE)
  set(OT_BACKBONE_ROUTER_MULTICAST_ROUTING ON CACHE STRING "" FORCE)
else()
  # Upstream requires BBR and its multicast routing together. Keep ordinary
  # Border Routing enabled above; do not advertise a nonfunctional BBR.
  set(OTBR_BACKBONE_ROUTER OFF CACHE BOOL "" FORCE)
  set(OT_BACKBONE_ROUTER_MULTICAST_ROUTING OFF CACHE STRING "" FORCE)
endif()
set(OTBR_OT_SRP_ADV_PROXY ON CACHE BOOL "" FORCE)
set(OTBR_OT_DISCOVERY_PROXY ON CACHE BOOL "" FORCE)
set(OTBR_SRP_SERVER_ON_INIT ON CACHE BOOL "" FORCE)
set(OTBR_SRP_SERVER_AUTO_ENABLE OFF CACHE BOOL "" FORCE)
set(OTBR_TREL OFF CACHE BOOL "" FORCE)
set(OTBR_NAT64 OFF CACHE BOOL "" FORCE)
set(OTBR_DHCP6_PD OFF CACHE BOOL "" FORCE)
set(OTBR_DNS_UPSTREAM_QUERY OFF CACHE BOOL "" FORCE)
set(OTBR_POWER_CALIBRATION OFF CACHE BOOL "" FORCE)
set(OTBR_NFTABLES OFF CACHE BOOL "" FORCE)
set(OTBR_NOTIFY_UPSTART OFF CACHE BOOL "" FORCE)
set(OT_FIREWALL OFF CACHE STRING "" FORCE)
set(OT_COMMISSIONER ON CACHE STRING "" FORCE)
if(OTBR_HOST_BUILD_RUNTIME)
  set(OT_PING_SENDER ON CACHE STRING "" FORCE)
  set(OT_NETDIAG_CLIENT ON CACHE STRING "" FORCE)
endif()
set(OT_SRP_SERVER ON CACHE STRING "" FORCE)
set(OT_POSIX_RCP_HDLC_BUS ON CACHE BOOL "" FORCE)
set(OT_POSIX_SETTINGS_PATH "\"/data/otbr/state\"" CACHE STRING "" FORCE)
set(OTBR_VENDOR_NAME XiaomiGateway2Port CACHE STRING "" FORCE)
set(OTBR_PRODUCT_NAME StandaloneOTBRHost CACHE STRING "" FORCE)
set(OTBR_VERSION mi-host-core-baseline CACHE STRING "" FORCE)
set(INSTALL_SYSTEMD_UNIT OFF CACHE BOOL "" FORCE)
set(OTBR_LAUNCHD_PLIST_DIR "" CACHE STRING "" FORCE)
set(OT_THREAD_VERSION 1.4 CACHE STRING "" FORCE)
add_subdirectory("${OTBR_HOST_ROOT}/vendor/ot-br-posix" otbr EXCLUDE_FROM_ALL)
# The upstream default (2) makes this gateway yield SRP publication in a LAN
# with several existing BRs. Keep ordinary unicast selection with a target of 8.
# This is a Network Data publication target, not the number of SRP clients.
# Share it with every core consumer; never edit the pinned vendor checkout.
target_compile_definitions(ot-config INTERFACE
  OPENTHREAD_CONFIG_NETDATA_PUBLISHER_DESIRED_NUM_UNICAST_DNS_SRP_SERVICE_ENTRIES=8)
# Stock Xiaomi rootfs has /var/run but no /run. Share this definition with the
# daemon and ot-ctl through their existing common configuration target.
target_compile_definitions(ot-posix-config INTERFACE
  "OPENTHREAD_POSIX_CONFIG_DAEMON_SOCKET_BASENAME=\"/var/run/openthread-%s\"")
if(OTBR_HOST_BUILD_RUNTIME)
  target_sources(openthread-posix PRIVATE "${OTBR_HOST_ROOT}/src/infra_adapter.cpp")
  target_include_directories(openthread-posix PRIVATE "${OTBR_HOST_ROOT}/include")
  include("${CMAKE_CURRENT_LIST_DIR}/commissioner_discovery.cmake")
  if(MI_OTBR_USERSPACE_MROUTE)
    include("${CMAKE_CURRENT_LIST_DIR}/bbr_userspace.cmake")
  endif()
endif()
